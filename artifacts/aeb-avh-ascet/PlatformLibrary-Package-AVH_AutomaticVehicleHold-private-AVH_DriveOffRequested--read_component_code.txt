// Purpose of this class is to determine a drive off request for AVH
// so far only driver drive off request is considered

if	(	(DriverAcceleratorApplied == true)		// Driver accelerator is pressed
	||	((ClutchEngagementPoint == true) && (	GearboxType == CSMGearboxType_ManualWoGearInfo
											||	GearboxType == CSMGearboxType_ManualWithGearInfo))    //Clutch is pressed and Gearbox is manual
	)
{
	 DriveOffRequested = true;
}
else
{
      DriveOffRequested = false;
}
