/*  This class prevents AVH from an immediate deactivation. A deactivation  */
/*  is only allowed after the driver has completely lifted the brake pedal. */

if  (   (AvhState == AVHState_Hold || LfcHoldActive == true)
     && BrakePedalIsApplied == false
     )	
{
    loc_DeactivationAllowedAfterActivation = true;
}
else if (   (AvhState == AVHState_NotActive
		  || AvhState == AVHState_MechHold
		  || AvhState == AVHState_DriveOff
		  || AvhState == AVHState_Release)
		  && LfcHoldActive == false
         )
{
    loc_DeactivationAllowedAfterActivation = false;
}
else
{
    // Do nothing
}
